```ts
import {
  convert,
  convertDetailed,
  reencode,
} from "@humanities/greek-conversion";

// Ancient Greek, from Beta Code as the TLG writes it
convert("*(/ELLHSIN E)GE/NETO", "beta-code", "greek");
// → Ἕλλησιν ἐγένετο

// Byzantine Greek, the καί symbol read with the manuscripts' accent
const res = convertDetailed("ϗ τὰ λοιπά", "greek", "transliteration", {
  greek: { kaiAccent: "byzantine" },
});
res.output; // → kaì tà loipá
res.losses[0].glyphVariant; // → kai-symbol
res.losses[1].diacritic; // → grave

// Modern Greek, from polytonic to monotonic, as passports write it
reencode("Hi ’Athína ’eînai hi protévousa", "transliteration", {
  preset: "iso-843-type-2",
  greek: { accentuation: "monotonic" },
});
// → I Athína eínai i protévousa
```
